<script>
// ========== 标签切换 ==========
const btns = document.querySelectorAll('.tab-btn');
const tabs = document.querySelectorAll('.tab-content');
btns.forEach(b=>{
    b.onclick=()=>{
        const t = b.dataset.tab;
        btns.forEach(x=>x.classList.remove('active'));
        b.classList.add('active');
        tabs.forEach(x=>x.classList.remove('show'));
        document.getElementById(t).classList.add('show');
    }
})

// ========== 贪吃蛇游戏 ==========
const canvas = document.getElementById('snakeCanvas');
const ctx = canvas.getContext('2d');
const scoreEl = document.getElementById('score');
const highEl = document.getElementById('highScore');
const size = 20;
let snake = [{x:10,y:10}];
let dx=1, dy=0;
let food={x:15,y:15};
let score=0;
let highScore = localStorage.getItem('snakeHigh')||0;
let gameRun=true;
let timer;
highEl.innerText=highScore;

function draw(){
    ctx.fillStyle='#0f172a';
    ctx.fillRect(0,0,canvas.width,canvas.height);
    //蛇
    ctx.fillStyle='#22c55e';
    snake.forEach(s=>{
        ctx.fillRect(s.x*size, s.y*size, size-2, size-2);
    })
    //食物
    ctx.fillStyle='#ef4444';
    ctx.fillRect(food.x*size, food.y*size, size-2, size-2);
}
function update(){
    if(!gameRun) return;
    const head = {...snake[0]};
    head.x += dx;
    head.y += dy;
    //撞墙
    if(head.x<0||head.y<0||head.x>=canvas.width/size||head.y>=canvas.height/size){
        gameOver();
        return;
    }
    //撞自己
    for(let s of snake){
        if(s.x===head.x && s.y===head.y){
            gameOver();
            return;
        }
    }
    snake.unshift(head);
    if(head.x===food.x && head.y===food.y){
        score+=10;
        scoreEl.innerText=score;
        randomFood();
    }else{
        snake.pop();
    }
    draw();
}
function randomFood(){
    food={
        x:Math.floor(Math.random()*(canvas.width/size)),
        y:Math.floor(Math.random()*(canvas.height/size))
    }
}
function gameOver(){
    gameRun=false;
    clearInterval(timer);
    if(score>highScore){
        highScore=score;
        localStorage.setItem('snakeHigh',highScore);
        highEl.innerText=highScore;
    }
    alert('游戏结束！得分：'+score+'，按R重新开始');
}
function startGame(){
    snake=[{x:10,y:10}];
    dx=1;dy=0;score=0;
    scoreEl.innerText=0;
    gameRun=true;
    clearInterval(timer);
    timer=setInterval(update,120);
    draw();
}
document.addEventListener('keydown',e=>{
    if(e.key==='r'||e.key==='R'){startGame();return;}
    if(e.key===' '){gameRun=!gameRun;return;}
    if(!gameRun) return;
    switch(e.key){
        case 'ArrowUp':case 'w':case 'W': if(dy!==1){dx=0;dy=-1;}break;
        case 'ArrowDown':case 's':case 'S': if(dy!==-1){dx=0;dy=1;}break;
        case 'ArrowLeft':case 'a':case 'A': if(dx!==1){dx=-1;dy=0;}break;
        case 'ArrowRight':case 'd':case 'D': if(dx!==-1){dx=1;dy=0;}break;
    }
})
document.querySelectorAll('.mobile-ctrl button').forEach(btn=>{
    btn.onclick=()=>{
        const d=btn.dataset.dir;
        if(d==='pause'){gameRun=!gameRun;return;}
        if(!gameRun) return;
        if(d==='up'&&dy!==1){dx=0;dy=-1;}
        if(d==='down'&&dy!==-1){dx=0;dy=1;}
        if(d==='left'&&dx!==1){dx=-1;dy=0;}
        if(d==='right'&&dx!==-1){dx=1;dy=0;}
    }
})
startGame();

// ========== 二维码模块 ==========
const qrCanvas = document.getElementById('qrCanvas');
function genQr(){
    const val = document.getElementById('qrText').value.trim();
    if(!val){alert('请输入网址/文字');return;}
    QRCode.toCanvas(qrCanvas, val, {width:320}, err=>{
        if(err) console.error(err);
    })
}
function fillCurrentUrl(){
    document.getElementById('qrText').value = location.href;
}
function downloadQr(){
    const link = document.createElement('a');
    link.download = 'qrcode.png';
    link.href = qrCanvas.toDataURL('image/png');
    link.click();
}
